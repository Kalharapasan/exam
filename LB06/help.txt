# 1. Build a Docker image named "student-python" with tag "v1" using the Dockerfile in the current directory (.)
sudo docker build -t student-python:v1 .

# 2. List all locally downloaded and built Docker images
sudo docker images

# 3. Run an interactive terminal session inside a new container created from the "student-python:v1" image
sudo docker run -it student-python:v1

# 4. Update the package list on your Linux system to get info on the latest available software versions
sudo apt-get update

# 5. Upgrade all installed packages on your Linux system to their latest versions
sudo apt-get upgrade

# 6. Install the Docker engine package on Ubuntu/Debian
sudo apt install docker.io

# 7. Check the installed Docker client version
docker --version

# 8. List all locally available Docker images
sudo docker images

# 9. Download (pull) the official "httpd" (Apache HTTP Server) image from Docker Hub
sudo docker pull httpd

# 10. Explicitly pull the latest version of the "httpd" image from Docker Hub
sudo docker pull httpd:latest

# 11. Pull a specific version (2.4.57) of the "httpd" image
sudo docker pull httpd:2.4.57

# 12. Pull the latest official Ubuntu OS image from Docker Hub
sudo docker pull ubuntu

# 13. Pull a specific version/tag ("lunar-20230816") of the Ubuntu image
sudo docker pull ubuntu:lunar-20230816

# 14. Create and run a container in detached mode (-d) with interactive terminal capabilities (-it) with a custom name
sudo docker run -d -it --name [My_container_name] [image_name]

# 15. View detailed JSON configuration and metadata of a Docker image or container
sudo docker inspect [image_id_or_image_name]

# 16. Show the commit history and layer structure of a Docker image
sudo docker history [image_id_or_image_name]

# 17. Save a Docker image to a compressed TAR archive file (useful for offline backup or transfer)
sudo docker save [image_id_or_image_name] > [filename.tar]

# 18. Remove (delete) a local Docker image from your machine
sudo docker rmi [image_id_or_image_name]

# 19. Load/restore a Docker image from a saved TAR archive file
sudo docker load -i [filename.tar]

# 20. Run an Ubuntu container named "ubuntu-dev" in detached (-d) and interactive (-it) mode
sudo docker run -d -it --name ubuntu-dev ubuntu:latest

# 21. Open an interactive Bash shell inside the running "ubuntu-dev" container
sudo docker exec -it ubuntu-dev /bin/bash

# 22. Update the package lists inside the running container (run this inside the container shell)
apt-get update

# 23. Install Python 3 inside the running container (run this inside the container shell)
apt-get install python3

# 24. Create a new image from a container's current changes (creates an unnamed/untagged image)
sudo docker commit [container_id_or_container_name]

# 25. Create a new image named "release7" with tag "1.3" from a container's current changes
sudo docker commit [container_id_or_container_name] release7:1.3

# 26. Run a new container named "ubuntu-testing-env" using the newly committed "release7:1.3" image
sudo docker run -d -it --name ubuntu-testing-env release7:1.3

# 27. Open an interactive Bash terminal inside the "ubuntu-testing-env" container
sudo docker exec -it ubuntu-testing-env /bin/bash

# 28. Authenticate and log in to your Docker Hub account from the command line
sudo docker login

# 29. Tag an existing local image so it matches your Docker Hub username repository format
sudo docker tag [release:version] [username]/[release:version]

# 30. Upload (push) your tagged image to your Docker Hub repository
sudo docker push [username]/[release:version]
